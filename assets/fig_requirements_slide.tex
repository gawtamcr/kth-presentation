% Slide copy of fig_motivation_requirements.tex without the (a)-(c) panel labels.
% Motivation figure, version 2: one learned skill, several requirements.
%
% The load-and-bond task of Section~\ref{sec:intro-motivation}, drawn as a
% timeline with one row per execution. The thick bar in each row is the
% interval during which the component is seated and held.
%
% Panel (a) shows executions sampled from a model trained on demonstrations.
% Panels (b) and (c) show the same executions under two requirements that
% differ only in the adhesive: a fast-curing adhesive must be seated early and
% held long, a slow-curing one later and briefly. Each requirement accepts a
% different subset of the same executions. The figure shows the gap without
% showing the method: the demonstrations contain executions that suit either
% adhesive, but neither window nor duration is stated in them, so the
% requirement has to be stated separately to select among the executions.
%
% Colours follow lib/includes.tex: color1bg for samples of the learned model,
% color2bg for the requirement, color4bg for an accepted execution, color5bg
% (faded) for a rejected one.
%
% Standalone, compiles independently. Build:
%   lualatex fig_motivation_requirements.tex && cp fig_motivation_requirements.pdf ../
%
% Included in the thesis as (\label{fig:motivation}):
%   \includegraphics[width=\linewidth]{figures/FlowMatching/fig_motivation_requirements.pdf}
\documentclass[tikz,border=2mm]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,calc}
\usepackage{amsmath,amssymb}

% fonts, matched to kth/kth-fonts.tex
\usepackage{fontspec}
\defaultfontfeatures{Ligatures={TeX}}
\setmainfont{TeX Gyre Termes}
\setsansfont{TeX Gyre Heros}
\usepackage{mathtools}
\usepackage[warnings-off={mathtools-colon,mathtools-overbracket}]{unicode-math}
\setmathfont{TeX Gyre Pagella Math}

% colours, copied from lib/includes.tex
\definecolor{color1bg}{HTML}{1954a6}
\definecolor{color2bg}{HTML}{24a0d8}
\definecolor{color3bg}{HTML}{d85497}
\definecolor{color4bg}{HTML}{b0c92b}
\definecolor{color5bg}{HTML}{63666a}
\colorlet{samplecol}{color1bg!75}
\colorlet{okcol}{color4bg!70!black}
\colorlet{offcol}{color5bg!25}

% the executions sampled from the learned model: seat time / release time
\def\executions{2.6/4.2, 2.9/5.9, 3.3/4.6, 3.6/6.6, 3.9/7.0, 4.3/5.4,
                4.6/7.8, 5.0/6.2, 5.4/8.4, 5.8/7.6, 6.2/8.6}

% one execution row at height #1: seated from #2 to #3, in colour #4
\newcommand{\execution}[4]{%
  \draw[#4, line width=0.6pt] (0.2,#1) -- (#2,#1);
  \draw[#4, line width=0.6pt] (#3,#1) -- (9.8,#1);
  \draw[#4, line width=3.6pt, line cap=round] (#2,#1) -- (#3,#1);}

% common frame of a panel: time axis and title #1
\newcommand{\panelframe}[1]{%
  \draw[-{Latex[length=1.6mm]}, line width=0.6pt] (0,-0.1) -- (10.2,-0.1);
  \node[anchor=north east, font=\footnotesize, inner sep=2pt] at (10.2,-0.15) {time};
  \node[anchor=south west, align=left, inner sep=0pt] at (0,4.25) {#1};}

% a panel filtered by a requirement: window [#1, #2], hold at least #3, title #4, index #5
\newcommand{\requirementpanel}[5]{%
  \fill[color2bg!14] (#1,-0.1) rectangle (#2,3.95);
  \draw[color2bg, line width=0.5pt] (#1,-0.1) -- (#1,3.95) (#2,-0.1) -- (#2,3.95);
  \node[anchor=north, font=\footnotesize, inner sep=2pt] at (#1,-0.15) {$a_#5$};
  \node[anchor=north, font=\footnotesize, inner sep=2pt] at (#2,-0.15) {$b_#5$};
  \panelframe{#4}
  \foreach \s/\e [count=\i] in \executions {
    \pgfmathsetmacro{\y}{3.85-0.34*(\i-1)}
    \pgfmathtruncatemacro{\ok}{and(and(\s>=#1, \s<=#2), \e-\s>=#3)}
    \ifnum\ok=1 \execution{\y}{\s}{\e}{okcol}
    \else       \execution{\y}{\s}{\e}{offcol}\fi
  }}

\begin{document}
\begin{tikzpicture}[x=0.46cm, y=1cm, font=\small]

%% ── (a) the learned model, no requirement ──
\begin{scope}
  \panelframe{Samples from the learned model}
  \foreach \s/\e [count=\i] in \executions {
    \pgfmathsetmacro{\y}{3.85-0.34*(\i-1)}
    \execution{\y}{\s}{\e}{samplecol}
  }
\end{scope}

%% ── (b) fast-curing adhesive: seat early, hold long ──
\begin{scope}[xshift=5.6cm]
  \requirementpanel{3.0}{4.5}{2.5}
    {Fast-curing adhesive\\[-1pt]\footnotesize seat within $[a_1,b_1]$, hold at least $d_1$}{1}
\end{scope}

%% ── (c) slow-curing adhesive: seat later, hold briefly ──
\begin{scope}[xshift=11.2cm]
  \requirementpanel{5.0}{6.5}{1.5}
    {Slow-curing adhesive\\[-1pt]\footnotesize seat within $[a_2,b_2]$, hold at least $d_2$}{2}
\end{scope}

%% ── key, in centimetres rather than panel time units ──
\begin{scope}[x=1cm, shift={(0,-0.95)}]
  \draw[samplecol, line width=3.6pt, line cap=round] (0.1,0) -- (0.7,0);
  \node[anchor=west, font=\footnotesize] at (0.85,0) {component seated and held};
  \draw[okcol, line width=3.6pt, line cap=round] (5.7,0) -- (6.3,0);
  \node[anchor=west, font=\footnotesize] at (6.45,0) {accepted by the requirement};
  \draw[offcol, line width=3.6pt, line cap=round] (11.3,0) -- (11.9,0);
  \node[anchor=west, font=\footnotesize] at (12.05,0) {rejected by the requirement};
\end{scope}
\end{tikzpicture}
\end{document}
